# OpenAI Codex app-server SDK dependency
#
# The default backend imports this Python module:
#
#   from codex_app_server import AppServerConfig, Codex
#
# The module is provided by the OpenAI Codex open-source SDK:
#
#   https://github.com/openai/codex
#   package: openai-codex-app-server-sdk
#   source:  codex/sdk/python
#
# Local editable install used by Agentic-Verification:
#
#   git clone https://github.com/openai/codex ../codex
#   pip install -e ../codex/sdk/python
#   npm install -g @openai/codex  # or: brew install --cask codex
#   export CODEX_BIN="$(which codex)"
#
# The editable SDK install does not provide the `codex` binary. CODEX_BIN must
# point to a separately installed or built OpenAI Codex executable that supports
# `codex app-server`.
#
# Keep `--backend=codex` / `codex exec` as legacy only. It does not provide the
# app-server thread/turn/event contract that VeriAgent records in manifests.
